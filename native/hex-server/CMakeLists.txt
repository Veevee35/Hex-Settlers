cmake_minimum_required(VERSION 3.16)
project(hex_state_restore LANGUAGES CXX)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(hex_state_restore state_restore.cpp)
target_compile_features(hex_state_restore PRIVATE cxx_std_17)
target_link_libraries(hex_state_restore PRIVATE CURL::libcurl OpenSSL::Crypto nlohmann_json::nlohmann_json)

add_executable(hex_turn_replay turn_replay.cpp)
target_compile_features(hex_turn_replay PRIVATE cxx_std_17)
target_include_directories(hex_turn_replay PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/cuda-exact)
target_include_directories(hex_state_restore PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/cuda-exact)
target_link_libraries(hex_turn_replay PRIVATE nlohmann_json::nlohmann_json)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_setup_turn test_setup_turn.cpp)
  target_compile_features(test_setup_turn PRIVATE cxx_std_17)
  add_test(NAME setup_turn COMMAND test_setup_turn)
  add_executable(test_main_turn test_main_turn.cpp)
  target_compile_features(test_main_turn PRIVATE cxx_std_17)
  target_link_libraries(test_main_turn PRIVATE nlohmann_json::nlohmann_json)
  add_test(NAME main_turn COMMAND test_main_turn)
endif()
